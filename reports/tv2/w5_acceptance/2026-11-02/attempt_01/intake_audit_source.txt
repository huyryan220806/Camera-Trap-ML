def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def read(p): return json.loads(Path(p).read_bytes())
def jsonl(p): return [json.loads(x) for x in Path(p).read_text(encoding='utf-8').splitlines() if x.strip()]
def write(p,v):
 p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
 with p.open('x',encoding='utf-8',newline='\n') as f: json.dump(v,f,indent=2,ensure_ascii=False,allow_nan=False);f.write('\n')
def git(*a): return subprocess.check_output(['git','-C',str(ROOT),*a])
def same(a,b,tol=1e-10):
 if isinstance(a,(int,float)) and not isinstance(a,bool): return isinstance(b,(int,float)) and math.isclose(a,b,abs_tol=tol,rel_tol=0)
 if isinstance(a,dict): return isinstance(b,dict) and a.keys()==b.keys() and all(same(v,b[k],tol) for k,v in a.items())
 if isinstance(a,list): return isinstance(b,list) and len(a)==len(b) and all(same(x,y,tol) for x,y in zip(a,b))
 return a==b

 receipt=(ROOT/'configs/dinov3_lp_w5.local.protocol.json').read_bytes()

def audit():
 sys.path.insert(0,str(ROOT))
 from src.evaluation import w5_validation as w
 parents,crops,hashes=w.load_frozen_validation(ROOT)
 handoff=read(RUN/'handoff.json');protocol=read(RUN/'protocol.approved.json')
 w.verify_handoff(handoff,RUN/'best_model.pt',hashes,protocol,RUN)
 crop_pred=jsonl(RUN/'val_crops.jsonl');image_pred=jsonl(RUN/'val_images.jsonl')
 derived=w.aggregate(crop_pred,parents,crops,'DINOv3_LP_run01',CHECKPOINT)
 report=w.audit_predictions(derived,parents,crops,'DINOv3_LP_run01',CHECKPOINT)
 submitted=w.audit_predictions(image_pred,parents,crops,'DINOv3_LP_run01',CHECKPOINT)
 assert report==submitted
 by_id={x['image_id']:x for x in image_pred}
 maxmean=max(abs(x-y) for p in derived for x,y in zip(p['class_scores'],by_id[p['image_id']]['class_scores']))
 assert maxmean<=1e-10
 import jsonschema
 for rows,schema in [(crop_pred,'w5_crop_prediction.schema.json'),(image_pred,'w5_image_prediction.schema.json')]:
  validator=jsonschema.Draft202012Validator(read(ROOT/'contracts'/schema))
  for row in rows:validator.validate(row)
 validation=read(RUN/'validation.json')
 assert len(validation['crop_predictions'])==680 and len(validation['image_predictions'])==600
 detailed={x['crop_id']:x for x in validation['crop_predictions']}
 for row in crop_pred:
  r=detailed[row['crop_id']];assert r['image_id']==row['image_id'] and r['y_true']==row['ground_truth']
  assert r['probabilities']==row['class_scores']
  assert all(math.isfinite(v) for v in r['logits'])
  # The source emits FP32 torch softmax; independent FP64 softmax is diagnostic,
  # not a replacement for the actual emitted softmax vectors near ties.
 assert same(report['metrics'],handoff['validation_metrics'])
 for key in report['metrics']: assert math.isclose(report['metrics'][key],validation['metrics'][key],abs_tol=1e-10,rel_tol=0)
 assert report['confusion_matrix']==validation['metrics']['confusion_matrix']
 log=read(RUN/'train_log.json');assert [r['epoch'] for r in log]==list(range(1,17))
 best=-1.;best_epoch=0;bad=0
 for r in log:
  epoch=r['epoch'];expected_lr=.01*(1+math.cos(math.pi*(epoch-1)/50))/2
  assert math.isclose(r['lr'],expected_lr,rel_tol=0,abs_tol=1e-15)
  assert all(math.isfinite(r[k]) for k in ['train_loss','val_loss','val_macro_f1','val_accuracy','lr','seconds'])
  if r['val_macro_f1']>best:best=r['val_macro_f1'];best_epoch=epoch;bad=0
  else:bad+=1
  if epoch<16: assert bad<7
 assert best_epoch==9 and bad==7 and best==validation['metrics']['macro_f1']
 reload=read(RUN/'reload_20261102T065157804272Z.json')
 assert reload['passed'] and reload['fresh_process'] and reload['max_abs_logit_delta']==0
 assert sha(RUN/'reload_20261102T065157804272Z.predictions.json')==reload['predictions_sha256']
 assert read(RUN/'reload_20261102T065157804272Z.predictions.json')==validation
 b2_path=ROOT/'reports/tv2/w5_evidence/attempt_01/validation_02/b2_predictions.jsonl'
 b2=jsonl(b2_path);b2report=w.audit_predictions(b2,parents,crops,b2[0]['experiment_id'],w.B2_SHA256)
 paired=w.compare_predictions(b2,derived);prior=read(RUN/'tv2_audit/audit.json')
 assert same(prior['metrics'],report['metrics']) and prior['paired_outcomes']==paired
 assert json.loads((RUN/'tv2_audit/auditor.log').read_bytes())==prior
 assert sha(RUN/'tv2_audit/predictions.jsonl')==prior.get('output_sha256',{}).get('predictions',sha(RUN/'tv2_audit/predictions.jsonl'))
 prior_predictions=jsonl(RUN/'tv2_audit/predictions.jsonl');assert same(prior_predictions,derived)
 delta={k:report['metrics'][k]-b2report['metrics'][k] for k in report['metrics']}

 for r in derived:

   errors.append(dict(image_id=r['image_id'],ground_truth=r['ground_truth'],b2=a['predicted_class'],dino=r['predicted_class'],
    class_scores=r['class_scores'],crop_count=r['crop_count'],location=manifest[r['image_id']]['location']))
 write(EVID/'independent_metrics.json',dict(status='PASS',dino=report,b2=b2report,metric_delta=delta,per_class_comparison=per_class,
  paired_counts={k:len(v) for k,v in paired.items()},paired_image_ids=paired,max_abs_image_mean_vector_delta=maxmean,
  best_epoch=best_epoch,epochs_completed=len(log),early_stop_bad_epochs=bad,
  submitted_image_predictions_verified=True,schemas='PASS',tv3_reload_integrity='PASS_NOT_INDEPENDENT_MODEL_REPLAY',
  tv3_auditor_integrity='PASS_RECOMPUTED_LOCALLY',model_loaded=False))

 print('Independent metrics PASS',report['metrics'],'paired',{k:len(v) for k,v in paired.items()},flush=True)

  '--protocol',str(RUN/'protocol.approved.json'),'--checkpoint',str(RUN/'best_model.pt'),'--crop-predictions',str(RUN/'val_crops.jsonl'),
  '--b2-predictions',str(b2_path),'--output',str(out)]
 p=subprocess.run(cmd,cwd=ROOT,capture_output=True)
 (EVID/'tv2_evaluator.log').write_bytes(p.stdout+p.stderr)

   evaluator_sha256=sha(ROOT/'scripts/evaluate_w5_validation.py'),module_sha256=sha(ROOT/'src/evaluation/w5_validation.py')))
